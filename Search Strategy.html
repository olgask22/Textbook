        // --- View Switching Logic ---

        function switchView(view) {

            const pyramidView = document.getElementById('pyramid-view');

            const processView = document.getElementById('process-view');

            const btnPyramid = document.getElementById('btn-pyramid');

            const btnProcess = document.getElementById('btn-process');


            if (view === 'pyramid') {

                pyramidView.classList.remove('hidden');

                processView.classList.add('hidden');

                

                // Update Button Styles

                btnPyramid.className = "px-4 py-2 rounded-md text-sm font-medium transition-all bg-slate-800 text-white shadow";

                btnProcess.className = "px-4 py-2 rounded-md text-sm font-medium transition-all text-gray-500 hover:bg-gray-100";

            } else {

                pyramidView.classList.add('hidden');

                processView.classList.remove('hidden');


                // Update Button Styles

                btnProcess.className = "px-4 py-2 rounded-md text-sm font-medium transition-all bg-slate-800 text-white shadow";

                btnPyramid.className = "px-4 py-2 rounded-md text-sm font-medium transition-all text-gray-500 hover:bg-gray-100";

            }

        }

        function createElementsList(elements) {

            return elements.map((el, i) => `

                <div class="text-sm border-b border-gray-100 pb-1 last:border-0">

                    <span class="font-semibold text-gray-700">${el.label}:</span>

                    <span class="text-gray-600">${el.value}</span>

                </div>

            `).join('');

        }


        function createProcessElementsList(elements) {

             return elements.map((el, i) => `

                <li class="text-xs">

                    <div class="font-bold text-gray-700 mb-0.5">${el.label}</div>

                    <div class="text-gray-500 leading-snug">${el.value}</div>

                </li>

            `).join('');

        }


        function renderPyramid() {

            const container = document.getElementById('pyramid-content');

            // Reverse order for pyramid (Why at bottom)

            // Original: Why, What, Where, How

            // Pyramid: How (top), Where, What, Why (base)

            const pyramidOrder = [...sections].reverse();

            

            // Keep the line

            let html = '<div class="absolute top-0 bottom-0 w-1 bg-gray-200 z-0 hidden md:block"></div>';


            pyramidOrder.forEach((section, index) => {

                // Width calculation

                let widthClass = 'md:w-11/12'; // Default widest

                if (index === 0) widthClass = 'md:w-5/12'; // Top

                if (index === 1) widthClass = 'md:w-7/12';

                if (index === 2) widthClass = 'md:w-9/12';


                html += `

                    <div class="relative z-10 w-full ${widthClass} mb-4 transition-all duration-500">

                        <div class="bg-white rounded-xl shadow-md overflow-hidden border-l-8 ${section.borderColor} hover:shadow-lg transition-shadow">

                            <div class="p-5 md:p-6 flex flex-col md:flex-row gap-6 items-start md:items-center">

                                <!-- Icon -->

                                <div class="flex-shrink-0 flex items-center justify-center w-16 h-16 rounded-full ${section.lightColor} ${section.textColor}">

                                    <i data-lucide="${section.icon}" class="w-8 h-8"></i>

                                </div>

                                

                                <!-- Content -->

                                <div class="flex-grow w-full">

                                    <div class="flex items-center gap-3 mb-2">

                                        <h2 class="text-2xl font-bold ${section.textColor}">${section.title}</h2>

                                        <span class="text-sm font-medium text-gray-400 uppercase tracking-wide">

                                            ${section.subtitle}

                                        </span>

                                    </div>

                                    <p class="text-gray-600 mb-4 font-medium italic">

                                        ${section.summary}

                                    </p>

                                    <div class="grid grid-cols-1 sm:grid-cols-2 gap-x-8 gap-y-2">

                                        ${createElementsList(section.elements)}

                                    </div>

                                </div>

                            </div>

                            <!-- Footer Stripe -->

                            <div class="h-2 w-full ${section.color} opacity-20"></div>

                        </div>

                    </div>

                `;

            });

            container.innerHTML = html;

        }


        function renderProcess() {

            const container = document.getElementById('process-content');

            let html = '';


            sections.forEach((section, index) => {

                // Arrow Logic

                const showArrow = index < sections.length - 1;

                const arrowHtml = showArrow ? `

                    <div class="hidden lg:block absolute top-12 -right-3 z-10 text-gray-300">

                        <svg width="24" height="24" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">

                            <path d="M5 12h14" />

                            <path d="m12 5 7 7-7 7" />

                        </svg>

                    </div>` : '';


                html += `

                    <div class="relative group">

                        ${arrowHtml}

                        <div class="h-full bg-white rounded-xl shadow-sm border border-gray-100 overflow-hidden flex flex-col hover:shadow-md transition-all duration-300 transform hover:-translate-y-1">

                            <!-- Number Badge -->

                            <div class="absolute top-4 right-4 text-6xl font-black text-gray-50 opacity-50 z-0 select-none">

                                ${index + 1}

                            </div>


                            <!-- Header -->
